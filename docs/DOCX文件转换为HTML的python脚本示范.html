from docx import Document
import html

def docx_to_html(docx_path, html_path):
    # 打开 DOCX 文件
      doc = Document(docx_path)
    #doc = Document( r"C:\1\Google chrome谷歌浏览器启用多线程下载功能教程.docx")

    # 初始化 HTML 内容
    html_content = "<html><body>"

    # 遍历每个段落，将其转换为 HTML 格式
    for para in doc.paragraphs:
        html_content += "<p>" + html.escape(para.text) + "</p>"

    # 遍历每个表格，将其转换为 HTML 格式
    for table in doc.tables:
        html_content += "<table border='1'>"
        for row in table.rows:
            html_content += "<tr>"
            for cell in row.cells:
                html_content += f"<td>{html.escape(cell.text)}</td>"
            html_content += "</tr>"
        html_content += "</table>"

    html_content += "</body></html>"

    # 将 HTML 内容写入文件
    with open(html_path, "w", encoding="utf-8") as html_file:
        html_file.write(html_content)

    print(f"文件已成功转换为 HTML: {html_path}")

3. 说明

Document: python-docx 库用于打开 DOCX 文件。

html.escape(): 用于将文本中的特殊字符（如 <, >, & 等）转义为 HTML 实体，避免 HTML 标签错误。

段落转换: 将每个段落的文本转为 HTML 的 <p> 标签。

表格转换: 如果 DOCX 文件中有表格，将每个表格行和单元格转换为 <table>, <tr>, <td> 标签。

最后，生成的 HTML 内容会被写入指定路径的 .html 文件中。

4. 执行

运行这个脚本时，输入的 DOCX 文件将被转换为 HTML 文件，并保存到你指定的路径。

#!/usr/bin/env python3
"""
docx_to_html.py - CLI 版本（支持文字+图片）

功能:
- 支持命令行指定输入输出文件
- 支持拖拽 .docx 文件到脚本
- 正确处理文字和图片（嵌入 Base64）
- 支持批量转换

依赖:
    pip install mammoth
"""

import sys
from pathlib import Path
import base64

try:
    import mammoth
except ImportError:
    print("Missing dependency: mammoth. Install with: pip install mammoth")
    sys.exit(1)


def convert_docx_to_html(input_path: Path, output_path: Path):
    if not input_path.exists():
        print(f"Input file does not exist: {input_path}")
        return

    def convert_image(image):
        with image.open() as image_bytes:
            data = image_bytes.read()
        b64 = base64.b64encode(data).decode("ascii")
        return {"src": f"data:{image.content_type};base64,{b64}"}

    with open(input_path, "rb") as docx_file:
        result = mammoth.convert_to_html(docx_file, convert_image=mammoth.images.inline(convert_image))
        html = result.value

    with open(output_path, "w", encoding="utf-8") as fo:
        fo.write(html)

    print(f"Converted: {input_path} -> {output_path}")


def main():
    args = sys.argv[1:]
    if not args:
        print("Usage: python docx_to_html.py input.docx [output.html]")
        print("       You can drag & drop multiple .docx files onto this script")
        sys.exit(0)

    for arg in args:
        input_path = Path(arg)
        if input_path.suffix.lower() != ".docx":
            print(f"Skipped (not .docx): {input_path}")
            continue

        # 自动生成输出路径
        if len(args) == 2 and Path(args[1]).suffix.lower() == ".html" and input_path.exists():
            output_path = Path(args[1])
        else:
            output_path = input_path.with_suffix(".html")

        convert_docx_to_html(input_path, output_path)


if __name__ == '__main__':
    main()
