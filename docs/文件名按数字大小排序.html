#在power shell 下运行

#遍历指定文件夹及其所有子文件夹（递归）

#按文件名中数字（超长数字安全）排序

#替换 ++ 为 _ _，+ 为 _

#去掉文件名首尾空格及中间所有空格

#给文件名加 全局递增数字前缀（两位补齐）

#保留文件扩展名

#不会覆盖已有文件，跳过空文件名

#“C:\XXX”=文件路径

----------------------------------

$folder = "C:\XXX"

$counter = 1 # 全局递增数字前缀

# 递归遍历文件夹

Get-ChildItem -Path $folder -File -Recurse |

Sort-Object {

# 提取文件名中所有数字，超长数字安全，按字符串排序

($_.BaseName -replace '\D','')

} |

ForEach-Object {

$origFullPath = $_.FullName

$origName = $_.Name

$extension = $_.Extension

# 替换 ++ 和 +

$newName = $origName -replace '\+\+', '_ _'

$newName = $newName -replace '\+', '_'

# 去掉首尾空格

$newName = $newName.Trim()

# 去掉文件名中间所有空格

$newName = $newName -replace '\s+', ''

# 保留扩展名

$newBaseName = [System.IO.Path]::GetFileNameWithoutExtension($newName)

$newExtension = [System.IO.Path]::GetExtension($newName)

if (![string]::IsNullOrWhiteSpace($newBaseName)) {

# 给文件名前加全局递增数字前缀

$newFileName = "{0:D2}_{1}{2}" -f $counter, $newBaseName, $newExtension

# 仅当新名字与原名字不同才重命名

if ($newFileName -ne $origName) {

Rename-Item -LiteralPath $origFullPath -NewName $newFileName

} else {

Write-Host "跳过：$origName （没有变化）"

}

$counter++

} else {

Write-Host "跳过：$origName （替换后为空）"

}

}
